/
├── index.html                           // Базовый HTML-файл и точка входа. Здесь подключается скрипт Telegram WebApp.
├── src/
│   ├── main.jsx                         // Главный загрузчик React. Монтирует App в корневой DOM-элемент.
│   ├── App.jsx                          // Главный сборщик-компонент. Управляет слоями (Z-index), Кинематографическим режимом и связывает все хуки с UI.
│   ├── index.css                        // Настройки Tailwind, глобальные стили и @keyframes анимации (параллакс, вылетающий урон).
│   ├── api/
│   │   └── cloudApi.js                  // Сетевой слой. Отправляет и загружает сохранения из Yandex Cloud Functions, используя авторизацию ТГ.
│   ├── config/
│   │   ├── constants.js                 // Глобальные настройки игры (интервалы автосейва, тикрейт, URL бэкенда).
│   │   ├── planets.js                   // База данных планет (множители ХП, цвета для градиентов, требуемые а.е. для перелета).
│   │   └── achievements.js              // Баланс Delta Mode. Формулы и коэффициенты роста для целей и наград достижений.
│   ├── utils/
│   │   ├── math.js                      // Чистая математика. Логика генерации ХП блоков (с джиттером) и расчет стоимости апгрейдов.
│   │   ├── formatters.js                // Преобразователи текста. Форматирование больших чисел (1.5M, 2B) и перевод в а.е.
│   │   └── telegram.js                  // Адаптер Telegram. Безопасные вызовы Haptic Feedback (вибрации) и настройка цвета рамки ТГ.
│   ├── store/                           // Глобальные хранилища (Zustand)
│   │   ├── storage.js                   // Наш умный кэш-адаптер. Спасает телефон от микрофризов, сбрасывая данные на диск раз в 30 секунд.
│   │   ├── useGameStore.js              // Ресурсы и базовая прогрессия (монеты, ХП текущего блока, комбо, индекс планеты).
│   │   ├── useUpgradesStore.js          // Инвентарь. Хранит уровни апгрейдов и "на лету" вычисляет текущий урон и шанс крита.
│   │   └── useAchieveStore.js           // Мозг Delta Mode. Трекает прогресс, отлавливает левел-апы и отправляет уведомления.
│   ├── hooks/                           // Мускулы игры (Изолированная логика)
│   │   ├── useCombat.js                 // Боевой движок. Считает урон от клика, криты, комбо, и спавнит вылетающие циферки.
│   │   ├── useAutoClicker.js            // Движок Bobo. Раз в секунду бьет по блоку, если дрон прокачан.
│   │   ├── useSaveSystem.js             // Диспетчер сохранений. Управляет таймерами автосейва и синхронизацией с облаком при старте.
│   │   ├── useRandomEvents.js           // Спавнер. Рассчитывает, когда и откуда вылетят кометы и астероиды, и считает награду за их поимку.
│   │   └── useVoyager.js                // Астрономический таймер. Вычисляет реальную дистанцию аппаратов Вояджер на текущую миллисекунду.
│   ├── components/
│   │   ├── layout/                      // Элементы интерфейса
│   │   │   ├── Header.jsx               // Шапка. Показывает текущую планету, шкалу полета к следующей, баланс и множитель комбо.
│   │   │   ├── Footer.jsx               // Подвал-магазин. Выводит кнопки прокачки модулей корабля.
│   │   │   └── Notifications.jsx        // Рендерер уведомлений. Анимирует появление и исчезновение карточек с полученными достижениями.
│   │   ├── panels/
│   │   │   └── AchievementsPanel.jsx    // Модальное окно. Сетка со всеми достижениями текущей планеты и их прогресс-барами.
│   │   └── arena/                       // Визуальная и интерактивная часть игры
│   │       ├── Background.jsx           // Звездное небо. Трехслойный параллакс на Canvas для эффекта скорости и полета.
│   │       ├── Block.jsx                // Основная мишень. Меняет трещины в зависимости от ХП, реагирует на тапы (onPointerDown).
│   │       ├── FloatingDamage.jsx       // Контейнер, который рисует разлетающиеся цифры урона с критами поверх блока.
│   │       ├── RandomEventsOverlay.jsx  // Слой, на котором физически пролетают астероиды, и по которым можно кликнуть.
│   │       ├── VoyagerHUD.jsx           // Атмосферная панель слева вверху с бегущими цифрами телеметрии зондов.
│   │       └── BoboDrone.jsx            // Компонент, анимирующий полет дрона Bobo вокруг блока по орбите.
│   └── assets/
│       └── images/                      // Статические медиафайлы
│           └── bobo-drone.webp          // Твое сгенерированное, оптимизированное (webp) изображение дрона без фона.
